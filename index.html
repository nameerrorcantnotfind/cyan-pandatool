CREATE TABLE fans (id bigint generated always as identity primary key,
  username text unique not null, display_name text, call_as text,
  intimacy int default 5, time_slot text, ai_style text, notes text,
  country text, category text, vip boolean default false, priority text default 'vừa',
  created_at timestamptz default now());
CREATE TABLE messages (id bigint generated always as identity primary key,
  fan_username text, direction text, content text, created_at timestamptz default now());
CREATE TABLE accounts (id bigint generated always as identity primary key,
  type text, username text, meta jsonb default '{}', created_at timestamptz default now());
CREATE TABLE logs (id bigint generated always as identity primary key,
  action text, result text, code text, message text, created_at timestamptz default now());
CREATE TABLE queue (id bigint generated always as identity primary key,
  fan_username text, content text, status text default 'pending', created_at timestamptz default now());
